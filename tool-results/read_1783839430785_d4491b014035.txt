   131→export function ProvidersPanel() {
   132→  const qc = useQueryClient();
   133→  const [addOpen, setAddOpen] = useState(false);
   134→  const [editing, setEditing] = useState<ProviderListItem | null>(null);
   135→  const [detailId, setDetailId] = useState<string | null>(null);
   136→
   137→  const { data, isLoading } = useQuery<{ providers: ProviderListItem[] }>({
   138→    queryKey: ["providers"],
   139→    queryFn: () => api.get("/api/providers"),
   140→  });
   141→
   142→  const delMut = useMutation({
   143→    mutationFn: (id: string) => api.del(`/api/providers/${id}`),
   144→    onSuccess: () => {
   145→      qc.invalidateQueries({ queryKey: ["providers"] });
   146→      qc.invalidateQueries({ queryKey: ["stats"] });
   147→      toast.success("تم حذف المزود");
   148→    },
   149→    onError: (e) => toast.error((e as Error).message),
   150→  });
   151→
   152→  return (
   153→    <div className="space-y-6">
   154→      <div className="flex items-center justify-between gap-3 flex-wrap">
   155→        <div>
   156→          <h1 className="text-2xl font-bold">المزودون</h1>
   157→          <p className="text-muted-foreground text-sm">
   158→            أضف عددًا غير محدود من المزودين والمفاتيح والـ Endpoints
   159→          </p>
   160→        </div>
   161→        <Button onClick={() => setAddOpen(true)}>
   162→          <Plus className="w-4 h-4 ml-2" />
   163→          إضافة مزود
   164→        </Button>
   165→      </div>
   166→
   167→      {isLoading ? (
   168→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   169→          {Array.from({ length: 3 }).map((_, i) => (
   170→            <Card key={i} className="animate-pulse">
   171→              <CardContent className="h-40" />
   172→            </Card>
   173→          ))}
   174→        </div>
   175→      ) : !data?.providers?.length ? (
   176→        <Card>
   177→          <CardContent className="py-16 flex flex-col items-center justify-center gap-3 text-center">
   178→            <div className="w-14 h-14 rounded-2xl bg-muted flex items-center justify-center">
   179→              <Server className="w-7 h-7 text-muted-foreground" />
   180→            </div>
   181→            <div className="font-semibold">لا يوجد مزودون بعد</div>
   182→            <p className="text-sm text-muted-foreground max-w-sm">
   183→              ابدأ بإضافة مزود الخدمة الأول (مثل OpenAI, Anthropic, Google…) مع
   184→              عنوانه الأساسي ومساراته.
   185→            </p>
   186→            <Button onClick={() => setAddOpen(true)} className="mt-2">
   187→              <Plus className="w-4 h-4 ml-2" />
   188→              إضافة مزود
   189→            </Button>
   190→          </CardContent>
   191→        </Card>
   192→      ) : (
   193→        <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
   194→          {data.providers.map((p) => (
   195→            <ProviderCard
   196→              key={p.id}
   197→              provider={p}
   198→              onOpen={() => setDetailId(p.id)}
   199→              onEdit={() => setEditing(p)}
   200→              onDelete={() => {
   201→                if (confirm(`حذف المزود "${p.name}" وكل مفاتيحه؟`)) {
   202→                  delMut.mutate(p.id);
   203→                }
   204→              }}
   205→            />
   206→          ))}
   207→        </div>
   208→      )}
   209→
   210→      <ProviderFormDialog
   211→        open={addOpen}
   212→        onOpenChange={setAddOpen}
   213→        editing={null}
   214→      />
   215→      <ProviderFormDialog
   216→        open={!!editing}
   217→        onOpenChange={(o) => !o && setEditing(null)}
   218→        editing={editing}
   219→      />
   220→
   221→      <ProviderDetailSheet
   222→        providerId={detailId}
   223→        onOpenChange={(o) => !o && setDetailId(null)}
   224→      />

   328→function ProviderFormDialog({
   329→  open,
   330→  onOpenChange,
   331→  editing,
   332→}: {
   333→  open: boolean;
   334→  onOpenChange: (o: boolean) => void;
   335→  editing: ProviderListItem | null;
   336→}) {
   337→  const qc = useQueryClient();
   338→  const [name, setName] = useState("");
   339→  const [baseUrl, setBaseUrl] = useState("");
   340→  const [authScheme, setAuthScheme] = useState("bearer");
   341→  const [authHeader, setAuthHeader] = useState("Authorization");
   342→  const [protocol, setProtocol] = useState("transparent");
   343→  const [priority, setPriority] = useState("0");
   344→  const [endpoints, setEndpoints] = useState<Array<{ type: string; path: string }>>(ENDPOINT_PRESETS);
   345→  const [modelsText, setModelsText] = useState("");
   346→  const [loading, setLoading] = useState(false);
   347→
   348→  // Sync when opening
   349→  const lastEditingId = useState(editing?.id)[0];
   350→  if (open && editing && editing.id !== lastEditingId && lastEditingId !== undefined) {
   351→    // noop; values are set below
   352→  }
   353→
   354→  function resetForm() {
   355→    setName("");
   356→    setBaseUrl("");
   357→    setAuthScheme("bearer");
   358→    setAuthHeader("Authorization");
   359→    setProtocol("transparent");
   360→    setPriority("0");
   361→    setEndpoints(ENDPOINT_PRESETS);
   362→    setModelsText("");
   363→  }
   364→
   365→  function prepareForEdit() {
   366→    if (editing) {
   367→      setName(editing.name);
   368→      setBaseUrl(editing.baseUrl);
   369→      setAuthScheme(editing.authScheme);
   370→      setAuthHeader(editing.authHeader);
   371→      setProtocol(editing.protocol || "transparent");
   372→      setPriority(String(editing.priority));
   373→      setEndpoints(
   374→        editing.endpoints.length > 0
   375→          ? editing.endpoints.map((e) => ({ type: e.type, path: e.path }))
   376→          : ENDPOINT_PRESETS
   377→      );
   378→      setModelsText("");
   379→    } else {
   380→      resetForm();
   381→    }
   382→  }
   383→
   384→  async function handleSubmit() {
   385→    if (!name.trim() || !baseUrl.trim()) {
   386→      toast.error("الاسم وعنوان URL مطلوبان");
   387→      return;
   388→    }
   389→    setLoading(true);
   390→    try {
   391→      const models = modelsText
   392→        .split(/[\n,]/)
   393→        .map((m) => m.trim())
   394→        .filter(Boolean);
   395→      if (editing) {
   396→        await api.patch(`/api/providers/${editing.id}`, {
   397→          name: name.trim(),
   398→          baseUrl: baseUrl.trim(),
   399→          authScheme,
   400→          authHeader,
   401→          protocol,
   402→          priority: parseInt(priority) || 0,
   403→        });
   404→        toast.success("تم تحديث المزود");
   405→      } else {
   406→        await api.post("/api/providers", {
   407→          name: name.trim(),
   408→          baseUrl: baseUrl.trim(),
   409→          authScheme,
   410→          authHeader,
   411→          protocol,
   412→          priority: parseInt(priority) || 0,
   413→          endpoints,
   414→          models,
   415→        });
   416→        toast.success("تمت إضافة المزود");
   417→      }
   418→      qc.invalidateQueries({ queryKey: ["providers"] });
   419→      qc.invalidateQueries({ queryKey: ["stats"] });
   420→      onOpenChange(false);
   421→      if (!editing) resetForm();
   422→    } catch (e) {
   423→      toast.error((e as Error).message);
   424→    } finally {
   425→      setLoading(false);
   426→    }
   427→  }
   428→
   429→  return (
   430→    <Dialog
   431→      open={open}
   432→      onOpenChange={(o) => {
   433→        if (o) prepareForEdit();
   434→        onOpenChange(o);
   435→      }}
   436→    >
   437→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   438→        <DialogHeader>
   439→          <DialogTitle>{editing ? "تعديل المزود" : "إضافة مزود جديد"}</DialogTitle>
   440→          <DialogDescription>
   441→            {editing
   442→              ? "عدّل بيانات المزود. تُدار المسارات والنماذج من شاشة التفاصيل."
   443→              : "أدخل بيانات المزود. سيتم إعداد المسارات الشائعة تلقائيًا ويمكنك تعديلها لاحقًا."}
   444→          </DialogDescription>
   445→        </DialogHeader>
   446→
   447→        <div className="space-y-4 py-2">
   448→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   449→            <div className="space-y-2">
   450→              <Label>اسم المزود *</Label>
   451→              <Input
   452→                value={name}
   453→                onChange={(e) => setName(e.target.value)}
   454→                placeholder="مثال: OpenAI"
   455→              />
   456→            </div>
   457→            <div className="space-y-2">
   458→              <Label>الأولوية</Label>
   459→              <Input
   460→                type="number"
   461→                value={priority}
   462→                onChange={(e) => setPriority(e.target.value)}
   463→                placeholder="0"
   464→                dir="ltr"
   465→              />
   466→              <p className="text-xs text-muted-foreground">
   467→                الأعلى يُفضّل عند توفّر عدة مزودين للنموذج نفسه
   468→              </p>
   469→            </div>
   470→          </div>
   471→
   472→          <div className="space-y-2">
   473→            <Label>عنوان URL الأساسي *</Label>
   474→            <Input
   475→              value={baseUrl}
   476→              onChange={(e) => setBaseUrl(e.target.value)}
   477→              placeholder="https://api.openai.com"
   478→              dir="ltr"
   479→            />
   480→          </div>
   481→
   482→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   483→            <div className="space-y-2">
   484→              <Label>طريقة المصادقة</Label>
   485→              <Select value={authScheme} onValueChange={setAuthScheme}>
   486→                <SelectTrigger>
   487→                  <SelectValue />
   488→                </SelectTrigger>
   489→                <SelectContent>
   490→                  {AUTH_SCHEMES.map((s) => (
   491→                    <SelectItem key={s.value} value={s.value}>
   492→                      {s.label}
   493→                    </SelectItem>
   494→                  ))}
   495→                </SelectContent>
   496→              </Select>
   497→            </div>
   498→            <div className="space-y-2">
   499→              <Label>اسم الترويسة (Header)</Label>
   500→              <Input
   501→                value={authHeader}
   502→                onChange={(e) => setAuthHeader(e.target.value)}
   503→                dir="ltr"
   504→              />
   505→            </div>
   506→          </div>
   507→
   508→          <div className="space-y-2">
   509→            <Label>البروتوكول (Protocol Translation)</Label>
   510→            <Select value={protocol} onValueChange={setProtocol}>
   511→              <SelectTrigger>
   512→                <SelectValue />
   513→              </SelectTrigger>
   514→              <SelectContent>
   515→                {PROTOCOLS.map((p) => (
   516→                  <SelectItem key={p.value} value={p.value}>
   517→                    {p.label}
   518→                  </SelectItem>
   519→                ))}
   520→              </SelectContent>
   521→            </Select>
   522→            <p className="text-xs text-muted-foreground">
   523→              استخدم <b>Anthropic</b> إذا كان المزود يستخدم بنية Claude Messages API
   524→              (مثل Anthropic نفسها أو مزود موحد يدعم claude). تحوّل البوابة طلبات
   525→              OpenAI تلقائياً إلى بنية Anthropic وتعيد الرد بصيغة OpenAI.
   526→            </p>
   527→          </div>
   528→
   529→          {!editing && (
   530→            <>
   531→              <div className="space-y-2">
   532→                <div className="flex items-center justify-between">
   533→                  <Label>المسارات (Endpoints)</Label>
   534→                  <Button
   535→                    type="button"
   536→                    variant="ghost"
   537→                    size="sm"
   538→                    onClick={() => setEndpoints([...endpoints, { type: "custom", path: "" }])}
   539→                  >
   540→                    <Plus className="w-3.5 h-3.5 ml-1" />
   541→                    إضافة مسار
   542→                  </Button>
   543→                </div>
   544→                <div className="space-y-2 max-h-48 overflow-y-auto border rounded-lg p-2 bg-muted/30">
   545→                  {endpoints.map((ep, i) => (
   546→                    <div key={i} className="flex gap-2">
   547→                      <Input
   548→                        className="w-32"
   549→                        value={ep.type}
   550→                        onChange={(e) => {
   551→                          const next = [...endpoints];
   552→                          next[i] = { ...next[i], type: e.target.value };
   553→                          setEndpoints(next);
   554→                        }}
   555→                        placeholder="النوع"
   556→                        dir="ltr"
   557→                      />
   558→                      <Input
   559→                        className="flex-1"
   560→                        value={ep.path}
   561→                        onChange={(e) => {
   562→                          const next = [...endpoints];
   563→                          next[i] = { ...next[i], path: e.target.value };
   564→                          setEndpoints(next);
   565→                        }}
   566→                        placeholder="/v1/chat/completions"
   567→                        dir="ltr"
   568→                      />
   569→                      <Button
   570→                        type="button"
   571→                        variant="ghost"
   572→                        size="icon"

   691→function KeysManager({ providerId, keys }: { providerId: string; keys: KeyItem[] }) {

   693→  const [adding, setAdding] = useState(false);
   694→  const [newKey, setNewKey] = useState("");
   695→  const [newName, setNewName] = useState("");
   696→  const [bulkMode, setBulkMode] = useState(false);
   697→  const [bulkKeys, setBulkKeys] = useState("");
   698→  const [loading, setLoading] = useState(false);
   699→
   700→  const addMut = useMutation({
   701→    mutationFn: () =>
   702→      api.post(`/api/providers/${providerId}/keys`, { key: newKey, name: newName }),
   703→    onSuccess: () => {
   704→      qc.invalidateQueries({ queryKey: ["provider", providerId] });
   705→      qc.invalidateQueries({ queryKey: ["providers"] });
   706→      qc.invalidateQueries({ queryKey: ["stats"] });
   707→      setNewKey("");
   708→      setNewName("");
   709→      setAdding(false);
   710→      toast.success("تمت إضافة المفتاح");
   711→    },
   712→    onError: (e) => toast.error((e as Error).message),
   713→  });
   714→
   715→  const addBulkMut = useMutation({
   716→    mutationFn: async () => {
   717→      // Split by newlines, commas, or spaces; trim; deduplicate; drop empties
   718→      const list = Array.from(
   719→        new Set(
   720→          bulkKeys
   721→            .split(/[\n,\s]+/)
   722→            .map((k) => k.trim())
   723→            .filter(Boolean)
   724→        )
   725→      );
   726→      let ok = 0;
   727→      let fail = 0;
   728→      for (const k of list) {
   729→        try {
   730→          await api.post(`/api/providers/${providerId}/keys`, { key: k });
   731→          ok++;
   732→        } catch {
   733→          fail++;
   734→        }

   736→      return { ok, fail, total: list.length };
   737→    },

   739→      qc.invalidateQueries({ queryKey: ["provider", providerId] });
   740→      qc.invalidateQueries({ queryKey: ["providers"] });
   741→      qc.invalidateQueries({ queryKey: ["stats"] });
   742→      setBulkKeys("");

   745→        toast.success(`تمت إضافة ${ok} مفتاح بنجاح`);

   762→

   764→    mutationFn: ({ id, isActive }: { id: string; isActive: boolean }) =>

   766→    onSuccess: () => {
   767→      qc.invalidateQueries({ queryKey: ["provider", providerId] });
   768→    },

   774→      api.patch(`/api/providers/${providerId}/keys/${id}`, { reset: true }),

   776→      qc.invalidateQueries({ queryKey: ["provider", providerId] });

   779→    onError: (e) => toast.error((e as Error).message),
   780→  });

   785→      await Promise.all(
   786→        disabled.map((k) =>
   787→          api.patch(`/api/providers/${providerId}/keys/${k.id}`, { reset: true })
   788→        )
   789→      );
   790→    },
   791→    onSuccess: () => {
   792→      qc.invalidateQueries({ queryKey: ["provider", providerId] });
   793→      qc.invalidateQueries({ queryKey: ["providers"] });
   794→      toast.success("تمت إعادة تفعيل جميع المفاتيح");
   795→    },
   796→    onError: (e) => toast.error((e as Error).message),
   797→  });

   816→              variant="outline"
   817→              onClick={() => resetAllMut.mutate()}
   818→              disabled={resetAllMut.isPending}

   820→              <RefreshCw className="w-4 h-4 ml-1" />
   821→              إعادة تفعيل الكل
   822→            </Button>

  1051→function keyStatusMeta(status: string) {
  1052→  switch (status) {
  1053→    case "active":
  1054→      return { label: "نشط", className: "text-emerald-600", icon: <CheckCircle2 className="w-3 h-3 ml-1" /> };
  1055→    case "rate_limited":
  1056→      return { label: "حد المعدل", className: "text-amber-600", icon: <Clock className="w-3 h-3 ml-1" /> };
  1057→    case "quota_exceeded":
  1058→      return { label: "تجاوز الحصة", className: "text-orange-600", icon: <AlertCircle className="w-3 h-3 ml-1" /> };
  1059→    case "disabled":
  1060→      return { label: "معطّل", className: "text-destructive", icon: <XCircle className="w-3 h-3 ml-1" /> };
  1061→    case "error":
  1062→      return { label: "خطأ", className: "text-destructive", icon: <AlertCircle className="w-3 h-3 ml-1" /> };
  1063→    case "exhausted":
  1064→      return { label: "منتهي", className: "text-destructive", icon: <XCircle className="w-3 h-3 ml-1" /> };
  1065→    default:
  1066→      return { label: status, className: "", icon: null };
  1067→  }

  1320→function timeAgo(iso: string): string {
  1321→  const d = new Date(iso).getTime();
  1322→  const diff = Date.now() - d;
  1323→  const min = Math.floor(diff / 60000);
  1324→  if (min < 1) return "الآن";
  1325→  if (min < 60) return `قبل ${min} د`;
  1326→  const hr = Math.floor(min / 60);
  1327→  if (hr < 24) return `قبل ${hr} س`;
  1328→  const day = Math.floor(hr / 24);
  1329→  return `قبل ${day} يوم`;
  1330→}
